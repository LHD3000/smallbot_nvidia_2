cmake_minimum_required(VERSION 3.16)
project(aruco_det)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc calib3d aruco)
find_package(yaml-cpp REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Marker.msg" "msg/MarkerArray.msg"
  DEPENDENCIES std_msgs geometry_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(aruco_det_node src/aruco_det.cpp)
target_include_directories(aruco_det_node PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(aruco_det_node rclcpp sensor_msgs geometry_msgs cv_bridge tf2 tf2_ros)
target_link_libraries(aruco_det_node "${cpp_typesupport_target}" ${OpenCV_LIBRARIES} yaml-cpp)
install(TARGETS aruco_det_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/generate_marker.py scripts/subscribe_xyz.py scripts/view_image.py
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_pose_utils test/test_pose_utils.cpp)
  target_include_directories(test_pose_utils PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_compile_definitions(test_pose_utils PRIVATE TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/test")
  ament_target_dependencies(test_pose_utils geometry_msgs tf2)
  target_link_libraries(test_pose_utils ${OpenCV_LIBRARIES} yaml-cpp)
  ament_add_pytest_test(marker_assets test/test_marker_assets.py TIMEOUT 60)
  ament_add_pytest_test(pipeline test/test_pipeline.py TIMEOUT 90)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
