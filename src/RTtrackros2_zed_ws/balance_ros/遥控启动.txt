# ==========================================
# 安装 launch 文件夹
# ==========================================
install(DIRECTORY launch
  DESTINATION share/${PROJECT_NAME}
)

colcon build --packages-select balance_ros --symlink-install

# 4. 一键启动！
ros2 launch balance_ros joystick_control.launch.py
